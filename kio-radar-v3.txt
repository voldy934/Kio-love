(() => {
  "use strict";

  const APP_VERSION = 3;
  const STORAGE_KEY = "kio-radar-v3";
  const LEGACY_STORAGE_KEYS = ["kio-radar-v2", "kio-radar-v1"];

  const MAX_PROFILES = 5000;
  const MAX_IMPORT_BYTES = 2 * 1024 * 1024; // 2 Mo

  const VIEW_IDS = [
    "onboarding",
    "radar",
    "detail",
    "history",
    "favorites"
  ];

  const NAV_VIEWS = new Set([
    "radar",
    "history",
    "favorites"
  ]);

  const $ = (id) => document.getElementById(id);

  let state = loadState();
  let currentView = "onboarding";
  let currentId = null;

  function createDefaultState() {
    return {
      version: APP_VERSION,
      profiles: [],
      history: [],
      favorites: [],
      lastAction: null,
      settings: {
        onboardingDone: false,
        lastView: "radar"
      }
    };
  }

  function makeId() {
    if (
      typeof crypto !== "undefined" &&
      typeof crypto.randomUUID === "function"
    ) {
      return `kio-${crypto.randomUUID()}`;
    }

    return (
      "kio-" +
      Date.now() +
      "-" +
      Math.random().toString(36).slice(2, 12)
    );
  }

  function normalize(value) {
    let username = String(value || "").trim();

    if (!username) return null;

    username = username
      .replace(
        /^(?:https?:\/\/)?(?:(?:www|m)\.)?instagram\.com\//i,
        ""
      )
      .replace(/^@+/, "")
      .replace(/^\/+/, "")
      .split(/[/?#\s]/)[0]
      .trim()
      .toLowerCase();

    if (!username || username.length > 30) {
      return null;
    }

    const reservedPaths = new Set([
      "p",
      "reel",
      "reels",
      "stories",
      "explore",
      "accounts",
      "direct"
    ]);

    if (reservedPaths.has(username)) {
      return null;
    }

    if (!/^[a-z0-9._]+$/i.test(username)) {
      return null;
    }

    return username;
  }

  function getColor(username) {
    const palettes = [
      ["#ff2aa9", "#6d35ff"],
      ["#ff5b91", "#263dff"],
      ["#a747ff", "#1c9cff"],
      ["#ff2c71", "#7029ff"],
      ["#ff6bce", "#3b52ff"],
      ["#e32cff", "#00bfff"]
    ];

    let hash = 0;

    for (let i = 0; i < username.length; i++) {
      hash =
        username.charCodeAt(i) +
        ((hash << 5) - hash);
    }

    return palettes[
      Math.abs(hash) % palettes.length
    ];
  }

  function sanitizeProfile(raw) {
    if (!raw || typeof raw !== "object") {
      return null;
    }

    const username = normalize(raw.username);

    if (!username) {
      return null;
    }

    const [c1, c2] = getColor(username);

    return {
      id:
        typeof raw.id === "string" &&
        raw.id.trim()
          ? raw.id.trim()
          : makeId(),

      username,

      viewed: Boolean(raw.viewed),

      viewedAt:
        Number.isFinite(raw.viewedAt)
          ? raw.viewedAt
          : raw.viewed
          ? Number.isFinite(raw.createdAt)
            ? raw.createdAt
            : Date.now()
          : null,

      favorite: Boolean(raw.favorite),

      createdAt:
        Number.isFinite(raw.createdAt)
          ? raw.createdAt
          : Date.now(),

      openedInstagramAt:
        Number.isFinite(raw.openedInstagramAt)
          ? raw.openedInstagramAt
          : null,

      c1,
      c2
    };
  }

  function normalizeState(rawState) {
    const clean = createDefaultState();

    if (!rawState || typeof rawState !== "object") {
      return clean;
    }

    const seenUsernames = new Set();
    const usedIds = new Set();

    const rawProfiles = Array.isArray(
      rawState.profiles
    )
      ? rawState.profiles
      : [];

    clean.profiles = rawProfiles
      .map(sanitizeProfile)
      .filter(Boolean)
      .filter((profile) => {
        if (
          seenUsernames.has(profile.username)
        ) {
          return false;
        }

        seenUsernames.add(profile.username);

        if (usedIds.has(profile.id)) {
          profile.id = makeId();
        }

        usedIds.add(profile.id);
        return true;
      })
      .slice(0, MAX_PROFILES);

    const ids = new Set(
      clean.profiles.map(
        (profile) => profile.id
      )
    );

    const legacyFavorites = new Set(
      Array.isArray(rawState.favorites)
        ? rawState.favorites.filter((id) =>
            ids.has(id)
          )
        : []
    );

    clean.profiles.forEach((profile) => {
      if (legacyFavorites.has(profile.id)) {
        profile.favorite = true;
      }
    });

    clean.favorites = clean.profiles
      .filter((profile) => profile.favorite)
      .map((profile) => profile.id);

    const history = Array.isArray(
      rawState.history
    )
      ? rawState.history
      : [];

    clean.history = Array.from(
      new Set(
        history.filter((id) =>
          ids.has(id)
        )
      )
    );

    clean.profiles.forEach((profile) => {
      if (
        profile.viewed &&
        !clean.history.includes(profile.id)
      ) {
        clean.history.push(profile.id);
      }
    });

    clean.lastAction =
      rawState.lastAction &&
      typeof rawState.lastAction ===
        "object" &&
      rawState.lastAction.type ===
        "markViewed" &&
      ids.has(
        rawState.lastAction.profileId
      )
        ? {
            type: "markViewed",
            profileId:
              rawState.lastAction.profileId
          }
        : null;

    const savedSettings =
      rawState.settings &&
      typeof rawState.settings ===
        "object"
        ? rawState.settings
        : {};

    const lastView =
      typeof savedSettings.lastView ===
        "string" &&
      NAV_VIEWS.has(
        savedSettings.lastView
      )
        ? savedSettings.lastView
        : "radar";

    clean.settings = {
      onboardingDone:
        Boolean(
          savedSettings.onboardingDone
        ) || clean.profiles.length > 0,
      lastView
    };

    return clean;
  }

  function loadState() {
    const keys = [
      STORAGE_KEY,
      ...LEGACY_STORAGE_KEYS
    ];

    for (const key of keys) {
      try {
        const raw =
          localStorage.getItem(key);

        if (!raw) continue;

        const parsed = JSON.parse(raw);

        return normalizeState(parsed);
      } catch (error) {
        console.warn(
          `KIO RADAR : lecture impossible pour ${key}`,
          error
        );
      }
    }

    return createDefaultState();
  }

  function syncDerivedState() {
    const ids = new Set(
      state.profiles.map(
        (profile) => profile.id
      )
    );

    state.favorites = state.profiles
      .filter((profile) => profile.favorite)
      .map((profile) => profile.id);

    state.history = Array.from(
      new Set(
        state.history.filter((id) =>
          ids.has(id)
        )
      )
    );

    state.profiles.forEach((profile) => {
      if (
        profile.viewed &&
        !state.history.includes(profile.id)
      ) {
        state.history.push(profile.id);
      }

      if (!profile.viewed) {
        state.history =
          state.history.filter(
            (id) => id !== profile.id
          );
      }
    });
  }

  function saveState() {
    syncDerivedState();

    state.version = APP_VERSION;

    try {
      localStorage.setItem(
        STORAGE_KEY,
        JSON.stringify(state)
      );
    } catch (error) {
      console.warn(
        "KIO RADAR : sauvegarde impossible",
        error
      );
    }

    updateStats();
    updateProgress();
  }

  function escapeHtml(value) {
    return String(value).replace(
      /[&<>"']/g,
      (char) => {
        const map = {
          "&": "&amp;",
          "<": "&lt;",
          ">": "&gt;",
          '"': "&quot;",
          "'": "&#039;"
        };

        return map[char];
      }
    );
  }

  function showToast(message) {
    const toast = $("toast");

    if (!toast) {
      console.log(message);
      return;
    }

    toast.textContent = message;
    toast.classList.add("show");

    clearTimeout(showToast.timer);

    showToast.timer = setTimeout(
      () => {
        toast.classList.remove("show");
      },
      1800
    );
  }

  function getProfile(id) {
    return state.profiles.find(
      (profile) => profile.id === id
    );
  }

  function parseProfileText(text) {
    return String(text || "")
      .split(/[\n,;\s]+/)
      .map((item) => item.trim())
      .filter(Boolean);
  }

  function addProfiles(values) {
    if (!Array.isArray(values)) {
      return;
    }

    const existing = new Set(
      state.profiles.map(
        (profile) => profile.username
      )
    );

    let added = 0;
    let invalid = 0;
    let duplicates = 0;
    let limitReached = false;

    for (const rawValue of values) {
      if (
        state.profiles.length >=
        MAX_PROFILES
      ) {
        limitReached = true;
        break;
      }

      const username =
        normalize(rawValue);

      if (!username) {
        invalid++;
        continue;
      }

      if (existing.has(username)) {
        duplicates++;
        continue;
      }

      const [c1, c2] =
        getColor(username);

      state.profiles.push({
        id: makeId(),
        username,
        viewed: false,
        viewedAt: null,
        favorite: false,
        createdAt: Date.now(),
        openedInstagramAt: null,
        c1,
        c2
      });

      existing.add(username);
      added++;
    }

    if (added > 0) {
      state.settings.onboardingDone = true;
    }

    saveState();
    renderAll();

    if (added > 0) {
      let message =
        `${added} profil` +
        `${added > 1 ? "s" : ""}` +
        ` ajouté` +
        `${added > 1 ? "s" : ""}`;

      if (duplicates > 0) {
        message +=
          ` · ${duplicates} doublon` +
          `${duplicates > 1 ? "s" : ""}`;
      }

      if (invalid > 0) {
        message +=
          ` · ${invalid} invalide` +
          `${invalid > 1 ? "s" : ""}`;
      }

      if (limitReached) {
        message += " · limite atteinte";
      }

      showToast(message);
      return;
    }

    if (limitReached) {
      showToast(
        `Limite de ${MAX_PROFILES} profils atteinte`
      );
    } else if (
      invalid > 0 &&
      duplicates === 0
    ) {
      showToast(
        "Aucun profil Instagram valide"
      );
    } else {
      showToast("Aucun nouveau profil");
    }
  }

  function markViewed(
    id,
    {
      rememberForUndo = true
    } = {}
  ) {
    const profile = getProfile(id);

    if (!profile) return false;

    const wasViewed = profile.viewed;

    profile.viewed = true;
    profile.viewedAt =
      profile.viewedAt || Date.now();

    state.history =
      state.history.filter(
        (item) => item !== id
      );

    state.history.unshift(id);

    if (!wasViewed && rememberForUndo) {
      state.lastAction = {
        type: "markViewed",
        profileId: id
      };
    }

    saveState();

    return !wasViewed;
  }

  function restoreProfile(
    id,
    {
      showMessage = true
    } = {}
  ) {
    const profile = getProfile(id);

    if (!profile) return false;

    profile.viewed = false;
    profile.viewedAt = null;

    state.history =
      state.history.filter(
        (item) => item !== id
      );

    if (
      state.lastAction &&
      state.lastAction.profileId === id
    ) {
      state.lastAction = null;
    }

    saveState();
    renderAll();

    if (showMessage) {
      showToast(
        "Profil remis dans Radar"
      );
    }

    return true;
  }

  function toggleFavorite(id) {
    const profile = getProfile(id);

    if (!profile) return;

    profile.favorite =
      !profile.favorite;

    saveState();
    renderAll();

    showToast(
      profile.favorite
        ? "Ajouté aux favoris"
        : "Retiré des favoris"
    );
  }

  function openInstagram(profile) {
    if (!profile) return;

    const url =
      "https://www.instagram.com/" +
      encodeURIComponent(
        profile.username
      ) +
      "/";

    // L'ouverture est faite directement
    // dans l'événement utilisateur pour limiter
    // les blocages de popup sur mobile.
    window.open(
      url,
      "_blank",
      "noopener,noreferrer"
    );

    profile.openedInstagramAt =
      Date.now();

    markViewed(profile.id);
    renderAll();
  }

  function createCard(profile) {
    const favoriteLabel =
      profile.favorite
        ? "Retirer des favoris"
        : "Ajouter aux favoris";

    return `
      <article
        class="profile-card"
        data-id="${escapeHtml(profile.id)}"
        tabindex="0"
        role="button"
        aria-label="Ouvrir le profil @${escapeHtml(
          profile.username
        )}"
      >
        <div
          class="profile-bg"
          style="--c1:${profile.c1};--c2:${profile.c2}"
        ></div>

        <div class="profile-noise"></div>

        <span class="status-chip">
          ${
            profile.viewed
              ? "VIEWED"
              : "UNSEEN"
          }
        </span>

        <button
          class="star ${
            profile.favorite
              ? "saved"
              : ""
          }"
          data-action="favorite"
          data-id="${escapeHtml(
            profile.id
          )}"
          aria-label="${favoriteLabel}"
          aria-pressed="${
            profile.favorite
              ? "true"
              : "false"
          }"
          type="button"
        >
          ${
            profile.favorite
              ? "★"
              : "☆"
          }
        </button>

        <div class="profile-info">
          <strong>
            @${escapeHtml(
              profile.username
            )}
          </strong>

          <small>
            Tap to open radar
          </small>
        </div>
      </article>
    `;
  }

  function renderGrid(
    element,
    profiles
  ) {
    if (!element) return;

    element.innerHTML =
      profiles.length > 0
        ? profiles
            .map(createCard)
            .join("")
        : "";
  }

  function getUnseen() {
    return state.profiles.filter(
      (profile) => !profile.viewed
    );
  }

  function renderRadar() {
    const grid = $("profileGrid");
    const empty = $("emptyRadar");
    const search = $("searchInput");

    if (!grid || !empty) return;

    const query = search
      ? search.value
          .trim()
          .toLowerCase()
          .replace(/^@/, "")
      : "";

    let profiles = getUnseen();

    if (query) {
      profiles = profiles.filter(
        (profile) =>
          profile.username.includes(
            query
          )
      );
    }

    renderGrid(grid, profiles);

    empty.classList.toggle(
      "hidden",
      profiles.length !== 0
    );
  }

  function renderHistory() {
    const grid = $("historyGrid");
    const empty = $("emptyHistory");
    const count = $("historyCount");

    if (!grid || !empty) return;

    const profiles = state.history
      .map(getProfile)
      .filter(Boolean);

    renderGrid(grid, profiles);

    empty.classList.toggle(
      "hidden",
      profiles.length !== 0
    );

    if (count) {
      count.textContent =
        profiles.length;
    }
  }

  function renderFavorites() {
    const grid = $("favoriteGrid");
    const empty = $("emptyFavorites");
    const count = $("favoriteCount");

    if (!grid || !empty) return;

    const profiles =
      state.favorites
        .map(getProfile)
        .filter(Boolean);

    renderGrid(grid, profiles);

    empty.classList.toggle(
      "hidden",
      profiles.length !== 0
    );

    if (count) {
      count.textContent =
        profiles.length;
    }
  }

  function renderDetail() {
    const container =
      $("detailCard");

    if (!container) return;

    const profile =
      getProfile(currentId);

    if (!profile) {
      showView("radar", {
        pushHistory: false
      });
      return;
    }

    container.innerHTML = `
      <div
        class="detail-visual"
        style="--c1:${profile.c1};--c2:${profile.c2}"
      >
        <div class="detail-content">

          <span class="kicker">
            ${
              profile.viewed
                ? "HISTORY"
                : "RADAR SIGNAL"
            }
          </span>

          <h2 class="detail-username">
            @${escapeHtml(
              profile.username
            )}
          </h2>

          <p class="detail-meta">
            Profil prêt à être ouvert sur Instagram.
          </p>

          <div class="detail-actions">

            <button
              class="primary-btn"
              id="openIg"
              type="button"
            >
              OPEN INSTAGRAM ↗
            </button>

            <button
              class="secondary-btn"
              id="detailFav"
              type="button"
              aria-pressed="${
                profile.favorite
                  ? "true"
                  : "false"
              }"
            >
              ${
                profile.favorite
                  ? "★ SAVED"
                  : "☆ SAVE"
              }
            </button>

          </div>

          ${
            profile.viewed
              ? `
                <button
                  class="secondary-btn full"
                  id="restoreBtn"
                  type="button"
                >
                  RESTORE TO RADAR
                </button>
              `
              : `
                <button
                  class="secondary-btn full"
                  id="markSeenBtn"
                  type="button"
                >
                  MARK AS VIEWED
                </button>
              `
          }

        </div>
      </div>
    `;

    $("openIg")?.addEventListener(
      "click",
      () => {
        openInstagram(profile);
      }
    );

    $("detailFav")?.addEventListener(
      "click",
      () => {
        toggleFavorite(profile.id);
      }
    );

    $("restoreBtn")?.addEventListener(
      "click",
      () => {
        restoreProfile(profile.id);
        showView("radar");
      }
    );

    $("markSeenBtn")?.addEventListener(
      "click",
      () => {
        markViewed(profile.id);
        renderAll();
        showView("history");
      }
    );
  }

  function updateProgress() {
    const percentElement =
      $("progressPercent");

    const ring =
      document.querySelector(
        ".progress-ring"
      );

    const total =
      state.profiles.length;

    const viewed =
      state.profiles.filter(
        (profile) =>
          profile.viewed
      ).length;

    const percent = total
      ? Math.round(
          (viewed / total) * 100
        )
      : 0;

    if (percentElement) {
      percentElement.textContent =
        `${percent}%`;
    }

    if (ring) {
      const degrees =
        Math.max(
          0,
          Math.min(
            360,
            percent * 3.6
          )
        );

      ring.style.background =
        `conic-gradient(
          #ff3bbf 0deg,
          #8b5cff ${degrees}deg,
          rgba(255,255,255,.08) ${degrees}deg,
          rgba(255,255,255,.08) 360deg
        )`;
    }
  }

  function updateStats() {
    const profiles =
      $("menuProfiles");

    const viewed =
      $("menuViewed");

    const saved =
      $("menuSaved");

    if (profiles) {
      profiles.textContent =
        state.profiles.length;
    }

    if (viewed) {
      viewed.textContent =
        state.profiles.filter(
          (profile) =>
            profile.viewed
        ).length;
    }

    if (saved) {
      saved.textContent =
        state.favorites.length;
    }
  }

  function renderAll() {
    renderRadar();
    renderHistory();
    renderFavorites();
    updateProgress();
    updateStats();

    if (
      currentView === "detail"
    ) {
      renderDetail();
    }
  }

  function updateBrowserHistory(
    view,
    replace = false
  ) {
    const historyState = {
      kioRadar: true,
      view,
      currentId:
        view === "detail"
          ? currentId
          : null
    };

    try {
      if (replace) {
        window.history.replaceState(
          historyState,
          ""
        );
      } else {
        window.history.pushState(
          historyState,
          ""
        );
      }
    } catch {
      // Navigation interne seulement
      // si History API est indisponible.
    }
  }

  function showView(
    view,
    {
      pushHistory = true,
      replaceHistory = false,
      scroll = true
    } = {}
  ) {
    if (
      !VIEW_IDS.includes(view) ||
      !$(view)
    ) {
      return;
    }

    if (
      view === "detail" &&
      !getProfile(currentId)
    ) {
      view = "radar";
      currentId = null;
    }

    currentView = view;

    VIEW_IDS.forEach((id) => {
      const section = $(id);

      if (section) {
        section.classList.add(
          "hidden"
        );
      }
    });

    $(view).classList.remove(
      "hidden"
    );

    const nav = $("bottomNav");

    if (nav) {
      nav.classList.toggle(
        "hidden",
        view === "onboarding" ||
          view === "detail"
      );
    }

    document
      .querySelectorAll(".nav-item")
      .forEach((button) => {
        button.classList.toggle(
          "active",
          button.dataset.view ===
            view
        );
      });

    if (NAV_VIEWS.has(view)) {
      state.settings.lastView =
        view;
      saveState();
    }

    if (view === "detail") {
      renderDetail();
    }

    if (pushHistory) {
      updateBrowserHistory(
        view,
        replaceHistory
      );
    }

    if (scroll) {
      window.scrollTo({
        top: 0,
        behavior: "smooth"
      });
    }
  }

  function closeMenu() {
    const menu = $("sideMenu");
    const menuButton =
      $("menuBtn");

    if (menu) {
      menu.classList.remove("open");
    }

    if (menuButton) {
      menuButton.setAttribute(
        "aria-expanded",
        "false"
      );
    }
  }

  function openMenu() {
    const menu = $("sideMenu");
    const menuButton =
      $("menuBtn");

    if (menu) {
      menu.classList.add("open");
    }

    if (menuButton) {
      menuButton.setAttribute(
        "aria-expanded",
        "true"
      );
    }
  }

  function openProfileCard(card) {
    if (!card) return;

    const id = card.dataset.id;

    if (!getProfile(id)) return;

    currentId = id;

    // IMPORTANT :
    // toucher la carte ne marque plus
    // automatiquement le profil comme vu.
    showView("detail");
  }

  function bindEvents() {
    $("enterBtn")?.addEventListener(
      "click",
      () => {
        state.settings.onboardingDone =
          true;

        saveState();
        showView("radar");
      }
    );

    document
      .querySelectorAll("[data-view]")
      .forEach((button) => {
        button.addEventListener(
          "click",
          () => {
            showView(
              button.dataset.view
            );
            closeMenu();
          }
        );
      });

    $("menuBtn")?.addEventListener(
      "click",
      () => {
        const menu =
          $("sideMenu");

        if (!menu) return;

        if (
          menu.classList.contains(
            "open"
          )
        ) {
          closeMenu();
        } else {
          openMenu();
        }
      }
    );

    $("closeMenu")?.addEventListener(
      "click",
      closeMenu
    );

    $("menuBackdrop")?.addEventListener(
      "click",
      closeMenu
    );

    document.addEventListener(
      "keydown",
      (event) => {
        if (event.key === "Escape") {
          closeMenu();
        }
      }
    );

    $("addProfilesBtn")?.addEventListener(
      "click",
      () => {
        const input =
          $("profileInput");

        if (!input) return;

        const text =
          input.value || "";

        if (!text.trim()) {
          showToast(
            "Ajoute au moins un profil"
          );
          return;
        }

        addProfiles(
          parseProfileText(text)
        );

        input.value = "";
      }
    );

    $("fileInput")?.addEventListener(
      "change",
      async (event) => {
        const input = event.target;

        const file =
          input.files &&
          input.files[0];

        if (!file) return;

        if (
          file.size >
          MAX_IMPORT_BYTES
        ) {
          showToast(
            "Fichier trop volumineux (2 Mo max)"
          );

          input.value = "";
          return;
        }

        try {
          const text =
            await file.text();

          addProfiles(
            parseProfileText(text)
          );
        } catch {
          showToast(
            "Impossible de lire ce fichier"
          );
        }

        input.value = "";
      }
    );

    $("searchInput")?.addEventListener(
      "input",
      renderRadar
    );

    [
      "profileGrid",
      "historyGrid",
      "favoriteGrid"
    ].forEach((gridId) => {
      const grid = $(gridId);

      if (!grid) return;

      grid.addEventListener(
        "click",
        (event) => {
          const favorite =
            event.target.closest(
              '[data-action="favorite"]'
            );

          if (favorite) {
            event.stopPropagation();

            toggleFavorite(
              favorite.dataset.id
            );

            return;
          }

          const card =
            event.target.closest(
              ".profile-card"
            );

          openProfileCard(card);
        }
      );

      grid.addEventListener(
        "keydown",
        (event) => {
          if (
            event.key !== "Enter" &&
            event.key !== " "
          ) {
            return;
          }

          const card =
            event.target.closest(
              ".profile-card"
            );

          if (!card) return;

          event.preventDefault();
          openProfileCard(card);
        }
      );
    });

    $("undoBtn")?.addEventListener(
      "click",
      () => {
        const action =
          state.lastAction;

        if (
          !action ||
          action.type !==
            "markViewed"
        ) {
          showToast(
            "Rien à annuler"
          );
          return;
        }

        const profile =
          getProfile(
            action.profileId
          );

        state.lastAction = null;

        if (!profile) {
          saveState();
          showToast(
            "Rien à annuler"
          );
          return;
        }

        profile.viewed = false;
        profile.viewedAt = null;

        state.history =
          state.history.filter(
            (id) =>
              id !== profile.id
          );

        saveState();
        renderAll();

        showToast(
          "Dernière action annulée"
        );

        showView("radar");
        closeMenu();
      }
    );

    $("resetBtn")?.addEventListener(
      "click",
      () => {
        const confirmed =
          window.confirm(
            "Supprimer toutes les données locales de KIO RADAR ?"
          );

        if (!confirmed) return;

        try {
          localStorage.removeItem(
            STORAGE_KEY
          );

          LEGACY_STORAGE_KEYS.forEach(
            (key) =>
              localStorage.removeItem(
                key
              )
          );
        } catch {}

        window.location.reload();
      }
    );

    window.addEventListener(
      "popstate",
      (event) => {
        const navState =
          event.state;

        if (
          navState &&
          navState.kioRadar
        ) {
          if (
            navState.view ===
            "detail"
          ) {
            currentId =
              navState.currentId ||
              currentId;
          }

          showView(navState.view, {
            pushHistory: false,
            scroll: false
          });

          closeMenu();
        }
      }
    );
  }

  function start() {
    bindEvents();

    // Sauvegarde immédiatement au format V3
    // si les données provenaient de V1/V2.
    saveState();

    const hasProfiles =
      state.profiles.length > 0;

    const initialView =
      hasProfiles
        ? state.settings.lastView
        : "onboarding";

    showView(initialView, {
      pushHistory: true,
      replaceHistory: true,
      scroll: false
    });

    renderAll();
  }

  if (
    document.readyState ===
    "loading"
  ) {
    document.addEventListener(
      "DOMContentLoaded",
      start,
      { once: true }
    );
  } else {
    start();
  }
})();
